import 'package:flutter/material.dart';
import 'package:flutter/services.dart';
import 'package:http/http.dart' as http;
import 'package:shared_preferences/shared_preferences.dart';
import 'dart:convert';
import 'dart:math';

class JWTSetupWizard extends StatefulWidget {
  @override
  _JWTSetupWizardState createState() => _JWTSetupWizardState();
}

class _JWTSetupWizardState extends State<JWTSetupWizard> {
  int currentStep = 0;
  bool isLoading = false;
  bool showPassword = false;

  final TextEditingController siteUrlController = TextEditingController();
  final TextEditingController usernameController = TextEditingController();
  final TextEditingController passwordController = TextEditingController();

  String generatedSecretKey = '';
  Map<String, dynamic> testResults = {};
  String? jwtToken;

  @override
  void initState() {
    super.initState();
    generatedSecretKey = _generateSecretKey();
  }

  String _generateSecretKey() {
    const chars = 'ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789!@#\$%^&*()_+-=[]{}|;:,.<>?';
    Random rnd = Random.secure();
    return String.fromCharCodes(Iterable.generate(64, (_) => chars.codeUnitAt(rnd.nextInt(chars.length))));
  }

  String _normalizeUrl(String url) {
    url = url.trim();
    if (url.isEmpty) return url;

    if (!url.startsWith('http://') && !url.startsWith('https://')) {
      url = 'https://$url';
    }

    if (url.endsWith('/')) {
      url = url.substring(0, url.length - 1);
    }

    return url;
  }

  Future<bool> _testWordPressConnection() async {
    try {
      final baseUrl = _normalizeUrl(siteUrlController.text);
      final testUrl = '$baseUrl/wp-json/wp/v2';

      print('Testing WordPress connection to: $testUrl');

      final response = await http.get(
        Uri.parse(testUrl),
        headers: {
          'User-Agent': 'Flutter-JWT-Setup/1.0',
          'Accept': 'application/json',
        },
      ).timeout(Duration(seconds: 15));

      print('WordPress API Response: ${response.statusCode}');

      if (response.statusCode == 200 || response.statusCode == 401) {
        return true;
      }

      return false;
    } catch (e) {
      print('WordPress connection error: $e');
      return false;
    }
  }

  Future<bool> _testJWTPlugin() async {
    try {
      final baseUrl = _normalizeUrl(siteUrlController.text);

      final response = await http.get(
        Uri.parse('$baseUrl/wp-json/'),
        headers: {'User-Agent': 'Flutter-JWT-Setup/1.0'},
      ).timeout(Duration(seconds: 10));

      if (response.statusCode == 200) {
        final routes = jsonDecode(response.body)['routes'];
        return routes != null && routes.containsKey('/jwt-auth/v1');
      }
      return false;
    } catch (e) {
      print('Error testing JWT plugin: $e');
      return false;
    }
  }

  Future<bool> _testJWTAuth() async {
    try {
      final baseUrl = _normalizeUrl(siteUrlController.text);
      final url = '$baseUrl/wp-json/jwt-auth/v1/token';

      final response = await http.post(
        Uri.parse(url),
        headers: {
          'Content-Type': 'application/json',
          'User-Agent': 'Flutter-JWT-Setup/1.0',
        },
        body: jsonEncode({
          'username': usernameController.text.trim(),
          'password': passwordController.text,
        }),
      ).timeout(Duration(seconds: 10));

      print('JWT Auth Response: ${response.statusCode}');
      print('JWT Auth Body: ${response.body}');

      if (response.statusCode == 200) {
        final data = jsonDecode(response.body);
        if (data['token'] != null) {
          jwtToken = data['token'];
          return true;
        } else {
          print('Token not found in response');
          return false;
        }
      } else {
        final error = jsonDecode(response.body);
        if (error.containsKey('message')) {
          print('JWT Error: ${error['message']}');
        }
        return false;
      }
    } catch (e) {
      print('JWT Auth error: $e');
      return false;
    }
  }

  Future<bool> _testTokenValidation() async {
    if (jwtToken == null) {
      print('Token not available for validation');
      return false;
    }

    try {
      final baseUrl = _normalizeUrl(siteUrlController.text);
      final url = '$baseUrl/wp-json/jwt-auth/v1/token/validate';

      final response = await http.post(
        Uri.parse(url),
        headers: {
          'Authorization': 'Bearer $jwtToken',
          'Content-Type': 'application/json',
          'User-Agent': 'Flutter-JWT-Setup/1.0',
        },
      ).timeout(Duration(seconds: 10));

      print('Token Validation Response: ${response.statusCode}');
      return response.statusCode == 200;
    } catch (e) {
      print('Token validation error: $e');
      return false;
    }
  }

  Future<void> _runAllTests() async {
    setState(() {
      isLoading = true;
      testResults.clear();
      jwtToken = null;
    });

    testResults['wordpress'] = await _testWordPressConnection();
    setState(() {});

    if (testResults['wordpress'] == true) {
      testResults['plugin'] = await _testJWTPlugin();
      setState(() {});
    }

    if (testResults['plugin'] == true) {
      testResults['jwt_auth'] = await _testJWTAuth();
      setState(() {});
    }

    if (testResults['jwt_auth'] == true && jwtToken != null) {
      testResults['token_validation'] = await _testTokenValidation();
      setState(() {});
    }

    setState(() {
      isLoading = false;
    });
  }

  bool _allTestsPassed() {
    return testResults['wordpress'] == true &&
           testResults['plugin'] == true &&
           testResults['jwt_auth'] == true;
  }

  void _nextStep() {
    if (currentStep < 3) {
      setState(() {
        currentStep++;
      });
    }
  }

  void _previousStep() {
    if (currentStep > 0) {
      setState(() {
        currentStep--;
      });
    }
  }

  void _copyToClipboard(String text) {
    Clipboard.setData(ClipboardData(text: text));
    ScaffoldMessenger.of(context).showSnackBar(
      SnackBar(
        content: Row(
          children: [
            Icon(Icons.check, color: Colors.white),
            SizedBox(width: 8),
            Text('Copiato negli appunti!'),
          ],
        ),
        backgroundColor: Colors.green,
        duration: Duration(seconds: 2),
      ),
    );
  }

  Future<String?> refreshJwtToken() async {
    try {
      final baseUrl = _normalizeUrl(siteUrlController.text);
      final url = '$baseUrl/wp-json/jwt-auth/v1/token';

      final response = await http.post(
        Uri.parse(url),
        headers: {
          'Content-Type': 'application/json',
          'User-Agent': 'Flutter-JWT-Setup/1.0',
        },
        body: jsonEncode({
          'username': usernameController.text.trim(),
          'password': passwordController.text,
        }),
      ).timeout(Duration(seconds: 10));

      if (response.statusCode == 200) {
        final data = jsonDecode(response.body);
        if (data['token'] != null) {
          jwtToken = data['token'];
          return jwtToken;
        }
      }
      return null;
    } catch (e) {
      print('Error refreshing token: $e');
      return null;
    }
  }

  void _saveConfiguration() async {
    final prefs = await SharedPreferences.getInstance();
    await prefs.setString('site_url', _normalizeUrl(siteUrlController.text));
    await prefs.setString('username', usernameController.text);
    await prefs.setString('jwt_secret_key', generatedSecretKey);
    await prefs.setString('jwt_token', jwtToken ?? '');

    ScaffoldMessenger.of(context).showSnackBar(
      SnackBar(
        content: Text('Configurazione salvata con successo!'),
        backgroundColor: Colors.green[600],
      ),
    );
  }

        currentStep: currentStep,

          if (step <= currentStep) {
            setState(() {
              currentStep = step;
            });
          }
        },

          return Padding(
            padding: const EdgeInsets.only(top: 16.0),

              children: [

                    onPressed: _previousStep,

                    onPressed: _getNextButtonAction(),

            ),
          );

          Step(
            title: Text('Credenziali WordPress'),

            title: Text('Verifica Configurazione'),

            title: Text('Installazione Plugin'),

            state: _getStepState(2),

            title: Text('Configurazione Finale'),
            content: _buildFinalStep(),
            isActive: currentStep >= 3,
            state: _getStepState(3),
          ),
        ],
      ),
    );
  }

  String _getWpConfigCode() {
    return '''// JWT Authentication Configuration
define('JWT_AUTH_SECRET_KEY', '$generatedSecretKey');
define('JWT_AUTH_CORS_ENABLE', true);
// Optional: Custom token expiration (default 7 days)
// define('JWT_AUTH_EXPIRE', time() + (DAY_IN_SECONDS * 30)); // 30 days''';
  }

  String _getHtaccessCode() {
    return '''# JWT Authentication - HTTP Authorization Header
RewriteEngine on
RewriteCond %{HTTP:Authorization} ^(.*)
RewriteRule ^(.*) - [E=HTTP_AUTHORIZATION:%1]
# For WPEngine hosting, use this instead:
# SetEnvIf Authorization "(.*)" HTTP_AUTHORIZATION=\$1''';
  }

  StepState _getStepState(int step) {
    if (currentStep > step) return StepState.complete;
    if (currentStep == step) return StepState.indexed;
    return StepState.disabled;
  }

  VoidCallback? _getNextButtonAction() {
    switch (currentStep) {
      case 0:
        return _isStep0Valid() ? _runAllTests : null;
      case 1:
        if (testResults.isNotEmpty && !isLoading) {
          return _nextStep;
        }
        return _runAllTests;
      case 2:
        if (testResults['plugin'] == true) {
          return _nextStep;
        } else {
          return () {
            setState(() {
              currentStep = 1;
            });
            _runAllTests();
          };
        }
      case 3:
        return () {
          Navigator.of(context).pop();
          _saveConfiguration();
        };
      default:
        return null;
    }
  }

  String _getNextButtonText() {
    switch (currentStep) {
      case 0:
        return 'Inizia Configurazione';
      case 1:
        if (isLoading) return 'Verifica in corso...';
        if (testResults.isEmpty) return 'Inizia Verifica';
        return 'Continua';
      case 2:
        return testResults['plugin'] == true ? 'Finalizza Setup' : 'Riprova Test';
      case 3:
        return 'Completa & Salva';
      default:
        return 'Avanti';
    }
  }

  Icon _getNextButtonIcon() {
    switch (currentStep) {
      case 0:
        return Icon(Icons.play_arrow);
      case 1:
        if (isLoading) return Icon(Icons.hourglass_empty);
        return Icon(Icons.search);
      case 2:
        return testResults['plugin'] == true ? Icon(Icons.arrow_forward) : Icon(Icons.refresh);
      case 3:
        return Icon(Icons.save);
      default:
        return Icon(Icons.arrow_forward);
    }
  }

  Color _getNextButtonColor() {
    switch (currentStep) {
      case 0:
        return _isStep0Valid() ? Colors.blue[600]! : Colors.grey;
      case 1:
        if (isLoading) return Colors.orange[600]!;
        if (_allTestsPassed()) return Colors.green[600]!;
        return Colors.blue[600]!;
      case 2:
        return testResults['plugin'] == true ? Colors.green[600]! : Colors.orange[600]!;
      case 3:
        return Colors.green[600]!;
      default:
        return Colors.blue[600]!;
    }
  }

  bool _isStep0Valid() {
    return siteUrlController.text.trim().isNotEmpty &&
           usernameController.text.trim().isNotEmpty &&
           passwordController.text.trim().isNotEmpty;
  }
